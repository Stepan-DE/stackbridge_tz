```python 3.12.3
def is_isomorphic(s: str, t: str) -> bool:  
    # Проверка одинаковой длины  
    if len(s) != len(t):  
        return False
    
    mapping = {}  # История замен  
    used_characters = set()  # Память для занятых букв второго слова  
    
    for i in range(len(s)):  
        char_s = s[i]  
        char_t = t[i]  
          
        # Если буква первого слова уже встречалась  
        if char_s in mapping:  
            # Проверяем, совпадает ли старое правило с текущей буквой  
            if mapping[char_s] != char_t:  
                return False
        
        # Если буквы первого слова еще не было  
        else:
            # Проверяем, не занята ли буква второго слова кем-то еще  
            if char_t in used_characters:  
                return False
        
		    # Если всё чисто, закрепляем правило  
		    mapping[char_s] = char_t  
		    used_characters.add(char_t)  
    return True
```

### Задание 2: Натуральная последовательность
```python 3.12.3
def missing_number(nums: list[int]) -> int:  
    # Так как одного числа не хватает прибавляем 1  
    n = len(nums) + 1  
    
    # Формула Гаусса для вычисления идеальной суммы от 1 до n  
    expected_sum = (n * (n + 1)) // 2  
    
    # Считаем реальную сумму чисел в списке  
    actual_sum = sum(nums)  
    
    # Находим разницу, результат и будет пропущенным числом  
    return expected_sum - actual_sum
    
nums = [1, 2, 3, 4, 5, 6, 8, 9, 10, 11] 
print(missing_number(nums))
```


Временная сложность O(N), так как мы используем функцию `sum(nums)`, которая один раз проходит по всему списку, чтобы сложить числа.

```python 3.12.3
def prime_factors(n: int) -> list[int]:  
    factors = []  # Список для хранения простых множителей  
    d = 2  # Минимальное простое число  
    
    # Ищем делители только до квадратного корня из n    
    while d * d <= n:  
        # Делим число n на d до тех пор, пока оно делится нацело  
        while n % d == 0:  
            factors.append(d)  
            n //= d  # Уменьшаем число n (целочисленное деление)  
        d += 1  # Переходим к следующему потенциальному делителю  
    
    # Если после циикла осталось число больше 1, то оно само является простым    
    if n > 1:  
        factors.append(n)  
    
    return factors
```

### Задание 2: Ручной счёт ROC_AUC
```python 3.12.3
import numpy as np  
  
  
def calculate_roc_auc_manual(y_true: np.ndarray, y_prob: np.ndarray) -> float:  
    """  
    Вычисляет метрику ROC-AUC векторным способом через вероятностный смысл.    """    # Выделяем вероятности для положительного и отрицательного классов  
    pos = y_prob[y_true == 1]  
    neg = y_prob[y_true == 0]  
  
    if len(pos) == 0 or len(neg) == 0:  
        raise ValueError("В данных должен присутствовать хотя бы один объект каждого класса.")  
  
    # Попарное сравнение через Broadcasting  
    correct_pairs = np.sum(pos[:, None] > neg) + 0.5 * np.sum(pos[:, None] == neg)  
    total_pairs = len(pos) * len(neg)  
  
    return float(correct_pairs / total_pairs)
```

### Задание 3: Ручной счёт корреляции
```python 3.12.3
import numpy as np  
  
  
def calculate_pearson_correlation(x: np.ndarray, y: np.ndarray) -> float:  
    """  
    Вычисляет коэффициент линейной корреляции Пирсона векторным способом.    """  
    if len(x) != len(y):  
        raise ValueError("Массивы x и y должны быть одинаковой длины.")  
    if len(x) < 2:  
        raise ValueError("Для расчета корреляции необходимо как минимум 2 элемента.")  
  
    # 1. Центрируем переменные (вычитаем среднее)  
    dev_x = x - np.mean(x)  
    dev_y = y - np.mean(y)  
  
    # 2. Считаем компоненты формулы через скалярное произведение и суммы квадратов  
    numerator = np.sum(dev_x * dev_y)  
    var_x = np.sum(dev_x ** 2)  
    var_y = np.sum(dev_y ** 2)  
  
    # 3. Проверка на пограничный случай (константные данные)  
    if var_x == 0 or var_y == 0:  
        raise ValueError("Корреляция не определена: один из массивов содержит константные значения (дисперсия = 0).")  
  
    denominator = np.sqrt(var_x * var_y)  
  
    return float(numerator / denominator)  
  
  
# ТЕСТЫ  
  
# 1. Данные из задания

print(f"1. Результат для тестового задания: {calculate_pearson_correlation(x_task, y_task):.2f}\n")
```
